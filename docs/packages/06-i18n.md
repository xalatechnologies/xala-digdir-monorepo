# @xala/i18n - Internationalization

@xala/i18n provides comprehensive internationalization support for the Xala Diglist Platform, enabling multi-language functionality with Norwegian (Bokmål) as primary and English as secondary language.

## Overview

The i18n system supports:
- **Multiple languages** with Norwegian (Bokmål) and English
- **Dynamic language switching** without page reload
- **Pluralization** and variable interpolation
- **Date/time formatting** per locale
- **Number formatting** with locale-specific rules
- **Namespace organization** for scalable translations

## Installation

```bash
# In your app directory
pnpm add @xala/i18n
```

## Quick Start

### 1. Setup Provider
```typescript
// app.tsx
import { I18nProvider } from '@xala/i18n';
import { translations } from './translations';

function App() {
  return (
    <I18nProvider 
      locale="nb" 
      translations={translations}
      fallback="en"
    >
      <Router>
        <Routes>
          {/* Routes */}
        </Routes>
      </Router>
    </I18nProvider>
  );
}
```

### 2. Use Translations
```typescript
// Component
import { useT } from '@xala/i18n';

function Greeting() {
  const t = useT();
  
  return (
    <div>
      <h1>{t('greeting.title')}</h1>
      <p>{t('greeting.subtitle', { name: 'John' })}</p>
    </div>
  );
}
```

### 3. Define Translations
```typescript
// translations/nb.ts
export const nb = {
  greeting: {
    title: 'Velkommen',
    subtitle: 'Hei, {name}!',
  },
  // ...
};

// translations/en.ts
export const en = {
  greeting: {
    title: 'Welcome',
    subtitle: 'Hello, {name}!',
  },
  // ...
};
```

## Core Features

### Translation Function
```typescript
const t = useT();

// Simple translation
t('button.save') // 'Lagre' or 'Save'

// With variables
t('user.greeting', { name: 'Jane' }) // 'Hei, Jane!' or 'Hello, Jane!'

// With count (pluralization)
t('item.count', { count: 1 }) // '1 element'
t('item.count', { count: 5 }) // '5 elementer'

// With fallback
t('missing.key', 'Default value') // 'Default value'
```

### Language Switching
```typescript
import { useLocale } from '@xala/i18n';

function LanguageSwitcher() {
  const { locale, setLocale } = useLocale();
  
  return (
    <Select value={locale} onValueChange={setLocale}>
      <Select.Trigger>
        <Select.Value />
      </Select.Trigger>
      <Select.Content>
        <Select.Item value="nb">Norsk</Select.Item>
        <Select.Item value="en">English</Select.Item>
      </Select.Content>
    </Select>
  );
}
```

### Date/Time Formatting
```typescript
import { useDateFormatter } from '@xala/i18n';

function DateDisplay({ date }: { date: Date }) {
  const { formatDate, formatTime, formatDateTime } = useDateFormatter();
  
  return (
    <div>
      <p>{formatDate(date)}</p> {/* '15.01.2024' or '01/15/2024' */}
      <p>{formatTime(date)}</p> {/* '14:30' or '2:30 PM' */}
      <p>{formatDateTime(date)}</p> {/* '15.01.2024 14:30' or '01/15/2024, 2:30 PM' */}
    </div>
  );
}
```

### Number Formatting
```typescript
import { useNumberFormatter } from '@xala/i18n';

function NumberDisplay({ value }: { value: number }) {
  const { formatNumber, formatCurrency, formatPercent } = useNumberFormatter();
  
  return (
    <div>
      <p>{formatNumber(1234.56)}</p> {/* '1 234,56' or '1,234.56' */}
      <p>{formatCurrency(1234.56, 'NOK')}</p> {/* 'kr 1 234,56' or 'NOK 1,234.56' */}
      <p>{formatPercent(0.75)}</p> {/* '75 %' or '75%' */}
    </div>
  );
}
```

## Advanced Features

### Pluralization Rules
```typescript
// Norwegian pluralization
const nbPluralization = {
  item: {
    zero: 'Ingen elementer',
    one: '1 element',
    two: '2 elementer',
    few: '{count} elementer',
    many: '{count} elementer',
    other: '{count} elementer',
  },
};

// Usage
t('item', { count: 0 }) // 'Ingen elementer'
t('item', { count: 1 }) // '1 element'
t('item', { count: 5 }) // '5 elementer'
```

### Context-Aware Translations
```typescript
// translations/nb.ts
export const nb = {
  button: {
    // Context based on usage
    save_form: 'Lagre skjema',
    save_changes: 'Lagre endringer',
    save_file: 'Lagre fil',
  },
};

// Usage with context
t('button.save', { context: 'form' }) // 'Lagre skjema'
t('button.save', { context: 'changes' }) // 'Lagre endringer'
```

### Rich Content
```typescript
// Support for HTML in translations
t('rich.content', {
  component: 'strong',
  children: 'Important text',
}) // '<strong>Important text</strong>'

// Support for components
t('welcome.message', {
  components: {
    name: <span className="highlight">{userName}</span>,
  },
}) // 'Welcome, <span class="highlight">John</span>!'
```

### Lazy Loading Translations
```typescript
// Load translations on demand
const loadTranslations = async (locale: string, namespace: string) => {
  const translations = await import(`./translations/${locale}/${namespace}.ts`);
  return translations.default;
};

// Usage in component
const [translations, setTranslations] = useState(null);

useEffect(() => {
  loadTranslations(locale, 'admin').then(setTranslations);
}, [locale]);
```

## Best Practices

### 1. Translation Keys
```typescript
// ✅ Good - Hierarchical and descriptive
'booking.form.title'
'validation.email.required'
'error.network.connection'

// ❌ Avoid - Flat or ambiguous
'title'
'email_error'
'network_error'
```

### 2. Variable Naming
```typescript
// ✅ Good - Descriptive variable names
const { t } = useT('booking');
const saveButtonText = t('actions.save');

// ❌ Avoid - Generic names
const text = t('save');
const str = t('cancel');
```

### 3. Default Values
```typescript
// ✅ Good - Provide meaningful defaults
t('optional.key', 'Default text')

// ❌ Avoid - Empty or unclear defaults
t('optional.key', '')
```

### 4. Namespace Usage
```typescript
// ✅ Good - Use namespaces for features
const t = useT('user.profile');
t('email') // 'user.profile.email'

// ✅ Also good - Full path
const t = useT();
t('user.profile.email')
```

## Translation Management

### Scanner Tool
```bash
# Scan for missing translations
node scripts/scan-i18n.js apps/web/src

# Output:
# Missing translations for nb:
# - listing.card.newFeature
# - booking.flow.step3

# Unused translations:
# - old.feature.title
# - deprecated.message
```

### Translation Workflow
1. **Add keys to code** using `t()` function
2. **Run scanner** to find missing keys
3. **Add translations** to locale files
4. **Review with team** for accuracy
5. **Test in UI** for proper display

### Version Control
```bash
# Track translation changes
git add packages/i18n/src/locales/
git commit -m "feat: add booking flow translations"

# Sync with translation team
git push origin feature/booking-translations
```

### Caching
```typescript
// Cache compiled translations
const translationCache = new Map();

function getTranslation(locale: string, key: string) {
  const cacheKey = `${locale}:${key}`;
  
  if (translationCache.has(cacheKey)) {
    return translationCache.get(cacheKey);
  }
  
  const translation = compileTranslation(locale, key);
  translationCache.set(cacheKey, translation);
  
  return translation;
}
```

## Testing

### Translation Testing
```typescript
// tests/i18n/translation.test.tsx
import { render, screen } from '@testing-library/react';
import { I18nProvider } from '@xala/i18n';
import { translations } from '../translations';

function renderWithTranslation(component, locale = 'nb') {
  return render(
    <I18nProvider locale={locale} translations={translations}>
      {component}
    </I18nProvider>
  );
}

test('renders translated text', () => {
  renderWithTranslation(<Greeting />);
  expect(screen.getByText('Velkommen')).toBeInTheDocument();
});

test('switches language', async () => {
  const { rerender } = renderWithTranslation(<Greeting />, 'nb');
  
  rerender(
    <I18nProvider locale="en" translations={translations}>
      <Greeting />
    </I18nProvider>
  );
  
  expect(screen.getByText('Welcome')).toBeInTheDocument();
});
```

## Troubleshooting

### Common Issues

#### Missing Translations
```typescript
// Check if key exists
const hasTranslation = useHasTranslation('key');
if (!hasTranslation) {
  console.warn('Missing translation:', 'key');
}
```

#### Performance Issues
```typescript
// Use memo for expensive translations
const expensiveTranslation = useMemo(() => {
  return t('complex.key', { complex: 'interpolation' });
}, [t, complexValue]);
```

#### Hydration Mismatch
```typescript
// Ensure server and client use same locale
const [locale, setLocale] = useState(initialLocale);
useEffect(() => {
  // Only update on client
  setLocale(navigator.language.startsWith('nb') ? 'nb' : 'en');
}, []);
```
